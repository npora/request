import { createClient } from '@npora/request/core'
import { retryPlugin } from
  '@npora/request/plugins/retry'

const api = createClient({ baseURL: '/api' })
  .use(retryPlugin({ retries: 2 }))

// Less boilerplate. More certainty.
interface User { id: number; name: string }
const user = await api.get<User>('/user')
console.log(user.name)

YOUR FIRST REQUEST

Install. Run. Meet your response.

  1. Install in a Node.js 22+ project.npm i @npora/request
  2. Save the example as first-request.mjs.
  3. Run it from the same project.node first-request.mjs

This Node.js example uses our public demo API. In a browser app, use your own same-origin /api or a backend that allows your origin through CORS.
